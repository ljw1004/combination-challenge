/** Deterministic geometry evidence; not native rendering or physical touch. */
import test from 'node:test';import assert from 'node:assert/strict';import {readFileSync} from 'node:fs';import * as M from '../src/elements-discovery-model.ts';
test('Elements desktop stays centered and no wider than Mars',()=>{const css=readFileSync(new URL('../src/elements-discovery.css',import.meta.url),'utf8');assert.match(css,/\.board\{max-width:600px;margin-inline:auto\}/);});
test('Birth is immediately beside target at phone and desktop widths',()=>{for(const width of [320,375,390,600]){const s=M.createNamed();const a=M.token(s,'earth')!,b=M.token(s,'water')!;a.x=18;a.y=30;b.x=50;b.y=30;const before=structuredClone(s);const r=M.stepNamed(s,{type:'combine',a:'earth',b:'water',boardWidth:width});assert(r.accepted);const c=M.token(r.state,'clay')!,card=Math.min(width*.28,110);assert(Math.abs(c.x-b.x)*width/100<=card+2+1e-7);assert(Math.abs(c.y-b.y)*10<=38+1e-7);assert(Math.abs(c.y-b.y)*10>=36||Math.abs(c.x-b.x)*width/100>=card,'A free immediate neighbour is used');for(const t of before.tokens)assert.deepEqual(M.token(r.state,t.id),t);}});
test('Crowding allows local overlap rather than a far-off free slot',()=>{const s=M.debugJump('comfort').state;for(let i=0;i<s.tokens.length;i++){s.tokens[i].x=18+(i%3)*32;s.tokens[i].y=30+Math.floor(i/3)*3.8;}const a=M.token(s,'flour')!,b=M.token(s,'air')!;b.x=50;b.y=37.6;const before=structuredClone(s);const r=M.stepNamed(s,{type:'combine',a:a.id,b:b.id,boardWidth:600});assert(r.accepted);const c=M.token(r.state,'starter')!;assert(Math.abs(c.x-b.x)*6<=112+1e-7);assert(Math.abs(c.y-b.y)*10<=38+1e-7);for(const t of before.tokens)assert.deepEqual(M.token(r.state,t.id),t);});
test('Old witness history cannot turn prior-stage objects gold after restore',()=>{const p=M.debugJump('world');assert(p.state.witnesses?.['home:light']);const restored=M.restore(JSON.stringify(p));assert(!M.victoryWitnesses(restored.state).has('window'));assert(!M.victoryWitnesses(restored.state).has('basket'));});

test('Cellar birth starts beside the rendered placed mixture, not the location center',()=>{let s=M.debugJump('comfort').state;s=M.stepNamed(s,{type:'combine',a:'flour',b:'air'}).state;s=M.stepNamed(s,{type:'place',token:'dough',location:'cellar'}).state;const dough=M.token(s,'dough')!;const r=M.stepNamed(s,{type:'combine',a:'starter',b:'dough',boardWidth:375});assert(r.accepted);assert.deepEqual(r.origin,{x:dough.x,y:dough.y+1.26});const c=M.token(r.state,'risen-dough')!;assert(Math.abs(c.y-r.origin!.y)<=3.8+1e-8);assert(Math.abs(c.x-r.origin!.x)*3.75<=107+1e-8);});

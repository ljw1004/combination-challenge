import test from 'node:test';
import assert from 'node:assert/strict';

const timer=(h:ReturnType<typeof locationHarness>)=>h.timers.filter(t=>t.ms===700).at(-1)?.fn();
const drag=(h:ReturnType<typeof locationHarness>,a:string,b:string)=>{const board=h.doc.querySelector('.board');board.onpointerdown({target:h.thing(a),button:0,clientX:52,clientY:100,pointerId:71,preventDefault(){}});h.setHit(h.thing(b));board.onpointermove({clientX:210,clientY:220,pointerId:71});board.onpointerup({clientX:210,clientY:220,pointerId:71});};
const advance=(h:ReturnType<typeof locationHarness>)=>h.click(h.button('chapter-next'));

test('one authored cue replaces legacy motivation/cache paragraphs and scene artwork is stable across same-place beats',()=>{
 const s=fixture(),h=locationHarness(s,undefined,{width:600});assert(h.button('mission').open);assert.equal(h.button('mission').dataset['cue-id'],'cue-0');assert(!h.app.innerHTML.includes(s.chapters[0].motivation));assert(!h.app.innerHTML.includes('All ordinary stock was here on arrival.'));assert.match(h.app.innerHTML,/class="board fixed-locations" style="height:844px"/);assert.equal(fromLegacyGame(h.state()).state.width,390);
 h.click(h.button('close'));const viewport:any=h.doc.querySelector('.viewport');viewport.scrollTop=231;const before=fromLegacyGame(h.state()).state.tokens;drag(h,'a','b');timer(h);advance(h);assert.equal(h.button('mission').dataset['cue-id'],'cue-1');assert.match(h.app.innerHTML,/data-background="A"/);h.click(h.button('close'));assert.equal((h.doc.querySelector('.viewport') as any).scrollTop,231);for(const t of before)assert.deepEqual(fromLegacyGame(h.state()).state.tokens.find(x=>x.id===t.id),t);
 const loaded=locationHarness(s,h.saved.get(h.key));assert(!loaded.button('mission').open);assert.equal(loaded.app.innerHTML.match(/data-background="A"/g)?.length,1);
});
test('fixed panels stack exactly and dragged storage move is atomic, bounded, saved, undoable and leaves chapter location alone',()=>{
 const s=fixture(),g=M.createGame(s);let game=g;for(const a of [{type:'combine',a:'a',b:'b'},{type:'advance'},{type:'combine',a:'c',b:'d'},{type:'advance'},{type:'combine',a:'e',b:'record'},{type:'advance'}] as const){const r=M.applyAction(s,game,a);assert(r.accepted,r.message);game=r.game;}
 const h=locationHarness(s,JSON.stringify({game,seen:[],tops:{A:0,B:844}}));h.click(h.button('close'));assert.match(h.app.innerHTML,/height:1688px/);assert.equal((JSON.parse(h.saved.get(h.key)!) as any).tops.B,844);
 const before=fromLegacyGame(h.state()),board:any=h.doc.querySelector('.board');board.onpointerdown({target:h.thing('f'),button:0,clientX:52,clientY:900,pointerId:1,preventDefault(){}});h.setHit(null);board.onpointermove({clientX:210,clientY:160,pointerId:1});board.onpointerup({clientX:210,clientY:160,pointerId:1});const moved=fromLegacyGame(h.state());assert.equal(moved.state.chapter,3);assert.equal(moved.state.storyLocation,'B');assert.equal(M.token(moved.state,'f')?.location,'A');assert.equal(moved.history.length,before.history.length+1);assert.equal(moved.events.at(-1)?.action.type,'relocate');assert.notEqual(M.arrivalCue(moved.state,'f'),'new');const loaded=locationHarness(s,h.saved.get(h.key));assert.equal(M.token(fromLegacyGame(loaded.state()).state,'f')?.location,'A');loaded.click(loaded.button('undo'));assert.deepEqual(fromLegacyGame(loaded.state()).state.tokens,before.state.tokens);
});

test('cross-panel target drops reject atomically, combine in one undo, and preserve a surviving dragged source at its origin',()=>{
 const s=fixture();let g=M.createGame(s);for(const a of [{type:'combine',a:'a',b:'b'},{type:'advance'},{type:'combine',a:'c',b:'d'},{type:'advance'},{type:'combine',a:'e',b:'record'},{type:'advance'}] as const)g=M.applyAction(s,g,a).game;
 g=M.applyAction(s,g,{type:'relocate',item:'g',location:'A',x:210,y:200}).game;
 const h=locationHarness(s,JSON.stringify({game:g,seen:[],tops:{A:0,B:844}}));h.click(h.button('close'));let before=fromLegacyGame(h.state()),board:any=h.doc.querySelector('.board');
 board.onpointerdown({target:h.thing('f'),button:0,clientX:52,clientY:950,pointerId:1,preventDefault(){}});h.setHit(h.thing('a'));board.onpointermove({clientX:52,clientY:100,pointerId:1});board.onpointerup({clientX:52,clientY:100,pointerId:1});let after=fromLegacyGame(h.state());assert.deepEqual(after.state.tokens,before.state.tokens);assert.equal(after.state.currentLocation,before.state.currentLocation);assert.equal(after.history.length,before.history.length);assert.equal(after.events.length,before.events.length+1);assert.equal(after.events.at(-1)?.accepted,false);
 h.flush();before=fromLegacyGame(h.state());board=h.doc.querySelector('.board');board.onpointerdown({target:h.thing('f'),button:0,clientX:52,clientY:950,pointerId:2,preventDefault(){}});h.setHit(h.thing('g'));board.onpointermove({clientX:210,clientY:200,pointerId:2});board.onpointerup({clientX:210,clientY:200,pointerId:2});after=fromLegacyGame(h.state());assert(M.token(after.state,'finish'));assert.equal(M.token(after.state,'finish')?.location,'A');assert.deepEqual(M.token(after.state,'f'),M.token(before.state,'f'));assert.equal(after.history.length,before.history.length+1);h.click(h.button('undo'));assert.deepEqual(fromLegacyGame(h.state()).state.tokens,before.state.tokens);
});

test('blocked source delivery warns once without an automatic retry loop, then retries after a real arrangement change',()=>{
 const s=fixture();let g=M.createGame(s);for(const a of [{type:'combine',a:'a',b:'b'},{type:'advance'},{type:'combine',a:'c',b:'d'},{type:'move',item:'kept',x:210,y:810}] as const){const r=M.applyAction(s,g,a);assert(r.accepted,r.message);g=r.game;}
 const h=locationHarness(s,JSON.stringify({game:g,seen:[],tops:{A:0}}));h.click(h.button('close'));h.flush();
 // Arrangement changes never advance; the user explicitly tries the arrow.
 let board:any=h.doc.querySelector('.board');board.onpointerdown({target:h.thing('kept'),button:0,clientX:210,clientY:810,pointerId:1,preventDefault(){}});h.setHit(null);board.onpointermove({clientX:200,clientY:800,pointerId:1});board.onpointerup({clientX:200,clientY:800,pointerId:1});timer(h);advance(h);
 const after=fromLegacyGame(h.state()),count=h.timers.filter(t=>t.ms===700).length;assert.equal(after.state.chapter,1);assert.equal(after.events.at(-1)?.accepted,false);assert.match((h.doc.querySelector('.feedback') as any).textContent,/needs room/);assert.equal((h.doc.querySelector('.feedback') as any).hidden,false);h.flush();assert.equal(h.timers.filter(t=>t.ms===700).length,count);assert.equal(fromLegacyGame(h.state()).events.length,after.events.length);
 board=h.doc.querySelector('.board');board.onpointerdown({target:h.thing('kept'),button:0,clientX:200,clientY:800,pointerId:2,preventDefault(){}});h.setHit(null);board.onpointermove({clientX:210,clientY:350,pointerId:2});board.onpointerup({clientX:210,clientY:350,pointerId:2});timer(h);advance(h);assert.equal(fromLegacyGame(h.state()).state.chapter,2);assert(fromLegacyGame(h.state()).state.arrivalLayout?.active);
});

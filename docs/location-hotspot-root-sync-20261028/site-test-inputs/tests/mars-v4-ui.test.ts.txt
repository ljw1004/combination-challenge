/** Historical Mars v4 through the current shared UI. Original click/automatic
 * handoff cases remain archived; all substantive goal/order/Undo assertions stay. */
import test from 'node:test';import assert from 'node:assert/strict';import {readFileSync} from 'node:fs';
import * as M from '../experiments/tangible-campaign/engine/model.ts';
import {encodeSave} from '../experiments/tangible-campaign/ui/save-codec.ts';
import {locationHarness} from '../experiments/tangible-campaign/ui/location-controller-harness.ts';
import {scenarios} from '../experiments/tangible-campaign/ui/pre-location-scenarios-v1.ts';
const marsV4=scenarios.find(s=>s.id==='mars-connected-survival-v4')!;
const v4Actions=JSON.parse(readFileSync(new URL('../experiments/tangible-campaign/evidence/mars-connected-survival-v4-route-evidence.json',import.meta.url),'utf8')).actionsLog;
function nightActions(buildFirst:boolean){let n=0;const actions=v4Actions.filter((a:any)=>{const here=n===9;if(a.type==='advance')n++;return here&&a.type!=='advance';});const construction=actions.filter((a:any)=>a.type==='combine'&&[a.a,a.b].some((id:string)=>['thermal-blanket','fitted-blanket','thermal-screen'].includes(id)));const experiment=actions.filter((a:any)=>!construction.includes(a));return buildFirst?[...construction,...experiment]:[...experiment,...construction];}
type H=ReturnType<typeof locationHarness>;
function start(){const g=M.checkpoint(marsV4,'chapter-10'),h=locationHarness(marsV4,encodeSave({game:g,tops:{},seen:[]}),{modernPointer:true});h.click(h.button('close'));return h;}
function drop(h:H,a:any){const source=a.type==='combine'?a.a:a.item,target=a.type==='combine'?a.b:'@'+a.location;assert(['combine','place'].includes(a.type));const priorEvents=h.state().events.length,src=h.thing(source),dst=h.thing(target);assert(src,source);assert(dst,target);h.advanceTime(500);const board=h.doc.querySelector('.board');board.onpointerdown({target:src,button:0,clientX:60,clientY:100,pointerId:1,preventDefault(){}});assert(dst.classList.contains('possible'));h.setHit(dst);board.onpointermove({clientX:210,clientY:250,pointerId:1});board.onpointerup({clientX:210,clientY:250,pointerId:1});const event=h.state().events.at(-1)!;assert.equal(event.accepted,true);assert(h.state().events.length>priorEvents,'The actual pointer gesture committed a new event');assert.deepEqual(event.action,a,'Historical non-location-plan worlds keep their combine/place event shape, not the newer drop envelope');}
function finishTimers(h:H){for(const t of h.timers.filter(t=>!t.ran&&!t.cancelled&&(t.ms===0||t.ms===700)))t.fn();}
for(const buildFirst of [false,true])test('Mars v4 current controller waits for both Night goals and then explicit arrow; buildFirst='+buildFirst,()=>{const h=start(),actions=nightActions(buildFirst);for(const [n,a]of actions.entries()){drop(h,a);assert.equal(h.state().state.stage,9);assert.equal(h.timers.filter(t=>t.ms===700).length,0,'No automatic chapter-travel timer: old v4 has no authored wonder here');if(n<actions.length-1){assert.equal(M.stageComplete(marsV4,h.state().state),false);assert(!h.button('chapter-next'));}}assert(M.stageComplete(marsV4,h.state().state));assert(M.token(h.state().state,'screened-cabin'));assert(M.token(h.state().state,'insulation-comparison'));finishTimers(h);assert.equal(h.state().state.stage,9);assert(!h.button('mission').open);const arrow=h.button('chapter-next');assert(arrow);h.click(arrow);assert.equal(h.state().state.stage,10);assert(h.button('mission').open);h.click(arrow);finishTimers(h);assert.equal(h.state().state.stage,10,'Repeated stale arrow cannot advance twice');});
test('Mars v4 Undo before explicit Night travel restores comparison progress and invalidates stale arrow',()=>{const h=start();for(const a of nightActions(true))drop(h,a);assert(M.stageComplete(marsV4,h.state().state));const arrow=h.button('chapter-next');assert(arrow);h.click(h.button('debug-toggle'));h.click(h.button('undo'));h.click(h.button('debug-close'));assert.equal(M.stageComplete(marsV4,h.state().state),false);assert(M.token(h.state().state,'screened-cabin'));assert(!M.token(h.state().state,'insulation-comparison'));assert(!h.button('chapter-next'));h.click(arrow);finishTimers(h);assert.equal(h.state().state.stage,9);assert(!h.button('mission').open);});

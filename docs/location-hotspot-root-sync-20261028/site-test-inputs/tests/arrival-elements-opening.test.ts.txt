// Logical/source tests only: no browser, DOM, controller or physical-interaction tests.
import {scenario as integrated} from '../experiments/tangible-campaign/ui/campaigns/elements-opening-wheel.ts';
import {scenario as previous} from '../experiments/tangible-campaign/ui/campaigns/elements.ts';
import test from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs';
import * as M from '../experiments/tangible-campaign/engine/model.ts';
import {validateContentLifecycles,validateObjectiveFlow,validateConstructionContinuity} from '../experiments/tangible-campaign/schema/lifecycle.ts';
import type {Scenario,Action} from '../experiments/tangible-campaign/schema/scenario.ts';
const root=new URL('../experiments/tangible-campaign/evidence/integration-20261028/sources/elements-opening-wheel-v1/',import.meta.url),s:Scenario=JSON.parse(fs.readFileSync(new URL('scenario.json',root),'utf8')),old:Scenario=JSON.parse(fs.readFileSync(new URL('baseline.scenario.json',root),'utf8'));
const output=new URL('../results/arrival-content-20261028/elements/',import.meta.url);fs.mkdirSync(output,{recursive:true});
assert.deepEqual(integrated,s);assert.deepEqual(previous,old);
const actions=s.checkpoints!.find(c=>c.id==='complete')!.actions;
const act=(g:M.Game,a:Action)=>{const r=M.applyAction(s,g,a);assert(r.accepted,`${JSON.stringify(a)}: ${r.message}`);return r;};
const combine=(a:string,b:string):Action=>({type:'combine',a,b});
const has=(g:M.Game,k:string)=>!!M.token(g.state,k);
const run=(a:Action[])=>{let g=M.createGame(s);const created=new Map<string,number>();const rows=s.stages.map((st,i)=>({number:i+1,id:st.id,earlierUsed:new Set<string>(),outputs:[] as string[],actions:[] as object[]}));for(const a0 of a){const stage=g.state.stage,r=act(g,a0);if(a0.type==='combine')for(const input of [a0.a,a0.b])if(created.has(input)&&created.get(input)!<stage)rows[stage].earlierUsed.add(input);if(r.created){created.set(r.created,stage);rows[stage].outputs.push(r.created);}rows[stage].actions.push({action:a0,created:r.created,retired:r.retired});g=r.game;g.history=[];}return {game:g,rows:rows.map(r=>({...r,earlierUsed:[...r.earlierUsed]}))};};
test('schema, lifecycle, objective and construction checks; no new supplies or mechanics',()=>{assert.deepEqual([...M.validateScenario(s),...validateContentLifecycles([s]),...validateObjectiveFlow([s]),...validateConstructionContinuity([s])],[]);assert.deepEqual(s.initialItems,old.initialItems);assert.deepEqual(s.items,old.items);assert.deepEqual(s.locations,old.locations);assert.deepEqual(s.stages.slice(1),old.stages.slice(1));assert.equal(s.recipes.length,old.recipes.length);assert.deepEqual(s.recipes.filter(r=>r.id!=='bowl'),old.recipes.filter(r=>r.id!=='bowl'));assert.equal(s.items.find(i=>i.id==='sealed-crock')!.label,'Lidded crock');});
test('first ten are nine unchanged opening recipes plus the existing fruit recipe',()=>{const r=run(actions);assert(r.game.state.completed);const created=r.rows.flatMap(r=>r.outputs);assert.deepEqual(created.slice(0,10),['clay','stone','brick','rain','plant','grain','flour','dough','flatbread','fruit']);const report={scope:'Author-aware exact logical route; not fresh play, human, browser or UI evidence.',...r,game:{completed:r.game.state.completed,metrics:r.game.state.metrics}};fs.writeFileSync(new URL('route-audit.json',output),JSON.stringify(report,null,2)+'\n');});
test('all 29 later chapters reuse earlier creations; chapter two requires opening Stone',()=>{const r=run(actions);for(const row of r.rows.slice(1))assert(row.earlierUsed.length,`${row.id} lacks earlier-created input`);assert.deepEqual(r.rows[1].earlierUsed,['stone']);assert(r.rows[2].earlierUsed.includes('brick'));assert.equal(r.rows.findIndex(r=>r.outputs.includes('bowl')),5);assert(r.rows[5].earlierUsed.includes('wheel'));});
test('all 31 checkpoints are earned prefixes and restore exactly; new identity does not overwrite old save',()=>{assert.notEqual(s.id,old.id);for(const cp of s.checkpoints!){assert.deepEqual(cp.actions,actions.slice(0,cp.actions.length));const g=M.checkpoint(s,cp.id);assert.deepEqual(M.restore(s,JSON.stringify(g)),g);}const g=M.checkpoint(old,'chapter-6');assert.deepEqual(M.restore(old,JSON.stringify(g)),g);assert.throws(()=>M.restore(s,JSON.stringify(g)));});
test('either opening objective order carries the made Stone into supper; no early bowl or kiln prerequisite',()=>{for(const route of [[combine('earth','water'),combine('clay','fire'),combine('earth','fire')],[combine('earth','fire'),combine('earth','water'),combine('clay','fire')]]){let g=M.createGame(s);for(const a of route)g=act(g,a).game;assert(M.stageComplete(s,g.state));assert(!has(g,'bowl'));assert(!has(g,'kiln'));assert(!has(g,'wheel'));g=act(g,{type:'advance'}).game;assert.equal(M.token(g.state,'stone')?.area,g.state.currentArea);for(const a of [combine('air','water'),combine('earth','rain'),combine('plant','air'),combine('grain','stone')])g=act(g,a).game;assert(has(g,'flour'));}});
test('normal highlighting exposes wheel at bowl-making; existing Wheel survives unchanged and undo is exact',()=>{const g=M.checkpoint(s,'chapter-6');const clay=M.token(g.state,'clay')!,wheel=M.token(g.state,'wheel')!;assert(clay&&wheel);assert.equal(clay.area,g.state.currentArea);assert.equal(wheel.area,g.state.currentArea);assert(M.partners(s,g.state,'clay').partners.includes('wheel'));assert(!M.partners(s,g.state,'clay').partners.includes('stone'));const r=act(g,combine('clay','wheel'));assert.equal(r.created,'bowl');assert.deepEqual(M.token(r.game.state,'wheel'),wheel);assert(!(r.retired??[]).includes('wheel'));const restored=M.undo(r.game).game;assert.deepEqual({...restored.state,metrics:g.state.metrics},g.state);assert.deepEqual(restored.history,g.history);});
test('all source copies survive until an authored combine/install event; chapter advance never retires objects',()=>{let g=M.createGame(s);for(const a of actions){const before=g.state.tokens.map(t=>t.id),r=act(g,a),after=new Set(r.game.state.tokens.map(t=>t.id)),removed=before.filter(id=>!after.has(id));if(a.type==='advance')assert.deepEqual(removed,[],`objects vanished on travel`);else for(const id of removed)assert(a.type==='combine'&&[a.a,a.b].includes(id),`${id} removed by unrelated event ${JSON.stringify(a)}`);g=r.game;g.history=[];}});

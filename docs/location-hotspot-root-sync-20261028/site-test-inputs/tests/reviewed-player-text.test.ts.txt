import test from 'node:test';import assert from 'node:assert/strict';
import {currentFamilies as before} from '../experiments/tangible-campaign/ui/stories/pre-reviewed-ux-content.ts';import {currentFamilies as after} from '../experiments/tangible-campaign/ui/stories/current-ux-content.ts';import {reviewedPlayerTextChanges,withReviewedPlayerText} from '../experiments/tangible-campaign/ui/stories/reviewed-player-text.ts';import * as M from '../experiments/tangible-campaign/engine/campaign.ts';
test('exact independent-review helper changes only254 explicit metadata fields across unchanged graph/session identities',()=>{assert.equal(reviewedPlayerTextChanges.length,254);for(const old of before){const raw=JSON.stringify(old),next=after.find(s=>s.id===old.id)!;assert.deepEqual(next,withReviewedPlayerText(old));assert.equal(JSON.stringify(old),raw);assert.deepEqual(next.recipes,old.recipes);assert.deepEqual(next.checkpoints,old.checkpoints);assert.deepEqual(M.validateScenario(next),[]);for(const i of old.items){const token={id:i.id,kind:i.id,x:100,y:100,location:old.chapters[0].location.id,hotspot:null,slots:{},incorporated:[]};assert(M.tokenHeight(next,token)<=M.tokenHeight(old,token),i.id+' non-growing footprint');}}});
test('exact review changes cannot silently bind to a different predecessor or alter Light objective counts',()=>{const d=structuredClone(before.find(s=>s.id==='divinity-final-reading-scenes-v7')!);d.items.find(i=>i.id==='nature-perfected-line')!.label+=' changed';assert.throws(()=>withReviewedPlayerText(d),/predecessor differs/);const light=after.find(s=>s.id==='divinity-light-reading-scenes-v4')!,old=before.find(s=>s.id===light.id)!;assert.equal(light.chapters.flatMap(c=>c.objectives).length,8);for(const c of light.chapters){assert.deepEqual(c.objectives.map(({text,...o})=>o),old.chapters.find(o=>o.id===c.id)!.objectives.map(({text,...o})=>o));assert(c.objectives.every(o=>o.text.endsWith('?')));}});

const accepted=(s:Scenario,g:M.Game,a:Action)=>{const r=M.applyAction(s,g,a);assert(r.accepted,`${s.id} ${JSON.stringify(a)}: ${r.message}`);return r.game;};

const local=(g:M.Game,id:string)=>id.startsWith('@')?g.state.places.some(p=>p.id===id.slice(1)&&p.area===g.state.currentArea):g.state.tokens.some(t=>t.id===id&&t.area===g.state.currentArea);
function prepare(s:Scenario,r:Recipe,width=375):{game:M.Game;prefix:Action[];suffix:Action[]}{
 if(s===workroom){
  const prefix:Action[]=[combine('papers','@press'),{type:'advance'}];
  if(r.a==='proof-back')prefix.push(combine('master','stock'),combine('block-readable','papers'));
  else prefix.push(combine('master','@window'),combine('reverse','stock'),combine('block-reflected','papers'));
  return {game:M.replay(s,prefix,width),prefix,suffix:[{type:'place',item:'block-readable',location:'display-frame'},{type:'advance'},{type:'place',item:'block-reflected',location:'printer-chest'},{type:'advance'}]};
 }
 const route=routeFor(s),prefix:Action[]=[];let g=M.createGame(s,width);
 for(let n=0;n<=route.length;n++){
  if(!g.state.discovered.includes(r.out!)&&local(g,r.a)&&local(g,r.b))return {game:g,prefix,suffix:route.slice(n)};
  assert(n<route.length,`No legal prefix for ${r.id}`);const a=route[n];
  if(match(s,a)?.out===r.out)continue; // Defer only the exact older result, never conjure its prerequisites.
  g=accepted(s,g,a);prefix.push(a);
 }
 throw Error('unreachable');
}
function finish(s:Scenario,g:M.Game,suffix:Action[]){
 for(const a of suffix){const r=match(s,a);if(r?.out&&g.state.discovered.includes(r.out))continue;g=accepted(s,g,a);}
 assert(g.state.completed);return g;
}

  assert.equal(expectedSourceHashAfterIntuitiveAlternatives(path,row.beforeSha256),row.afterSha256);
  assert.throws(()=>expectedSourceHashAfterIntuitiveAlternatives(path,'0'.repeat(64)),/exact LIVE86 predecessor/);
  assert.equal(sha(row.predecessorScenarioPath),row.predecessorScenarioSha256);
 }
 assert.equal(expectedSourceHashAfterIntuitiveAlternatives('unrelated','a'.repeat(64)),'a'.repeat(64));
});

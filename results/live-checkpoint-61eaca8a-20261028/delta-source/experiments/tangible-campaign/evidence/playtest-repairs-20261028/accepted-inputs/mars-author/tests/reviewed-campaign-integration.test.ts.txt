test('new scenarios are not grandfathered and every recipe has valid explicit input, objective and compound decisions',()=>{
 const acceptedScenarios=[...c85Scenarios,...intellectualWorldScenarios];
 for(const s of acceptedScenarios){assert(!archivedScenarioIds.has(s.id));for(const r of s.recipes)assert.deepEqual(validateRecipeLifecycle(r,true),[],s.id+':'+r.id);}

 const audit=lifecycleAudit;
 const priorAudit=JSON.parse(readFileSync(new URL('../experiments/tangible-campaign/reviews/lifecycle/audit.json',import.meta.url),'utf8'));

});
for(const spec of routeSpecs){const s=scenarioFor(spec.key),actions=routeActions(spec.source);
 test(`${s.id}: canonical current-goal ancestry, event-local input lifetime and preserved fitted contents at three widths`,()=>{
  const evidence=JSON.parse(readFileSync(new URL(`../experiments/tangible-campaign/evidence/${s.id}-route-evidence.json`,import.meta.url),'utf8'));assert.deepEqual(evidence.actionsLog,actions);
  for(const width of [320,375,600]){const result=auditRoute(s,actions,width);assert(result.game.state.completed);assert.equal(result.rows.length,s.stages.length);assert.deepEqual(result.missing,[]);assert.equal(result.game.state.metrics.combinations,evidence.combinations);assert.deepEqual(normalized(M.restore(s,JSON.stringify(result.game)).state),normalized(result.game.state));if(width===375){assert.deepEqual(result.rows,evidence.goalContinuity);assert.deepEqual(result.stageEntries.map(({stateSha256,...row})=>row),evidence.stageEntries.map(({stateSha256,...row}:any)=>row));}}
 });
 test(`${s.id}: every authored checkpoint restores and remains separate from every historical save identity`,()=>{
  for(const cp of s.checkpoints??[])for(const width of [320,375,600]){const g=M.checkpoint(s,cp.id,width);assert.deepEqual(normalized(M.restore(s,JSON.stringify(g)).state),normalized(g.state));for(const old of legacy)assert.throws(()=>M.restore(old,JSON.stringify(g)),/Invalid campaign save/);}
 });
 for(const alt of spec.alternatives)test(`${s.id}: documented ${alt.name} alternative completes with goal-connected reuse or explicitly earned early goals, and event-local behavior`,()=>{
  const actions=routeActions(alt.source,alt.witness),report=auditRoute(s,actions);assert(report.game.state.completed);
  if(spec.key!=='food'||!['early-kitchen-forward','early-kitchen-reverse'].includes(alt.name)){assert.deepEqual(report.missing,[]);return;}
  // These two explicit witnesses do the second chapter's real cooking early at the
  // same workstation. Do not invent fresh work or gates solely to satisfy an audit.
  const ready=['orchard','salads','pastry','sugar','custard','sponge','quickbread','meringue','precision'];
  assert.deepEqual(report.missing,[...ready,'snack']);
  const earned=acceptedGoalAncestry(s,actions);assert.equal(earned.rows.length,s.stages.length);assert.deepEqual(earned.rows.filter(r=>r.readyAtEntry).map(r=>r.stage),ready);
  for(const row of earned.rows.slice(1))assert(row.priorCreations.length,`${row.stage}: actual accepted ancestry must contain an earlier-created object`);
  for(const id of ready){const row=earned.rows.find(r=>r.stage===id)!;assert(row.readyAtEntry);assert.deepEqual(row.actions,[]);assert(row.roots.length);for(const root of row.roots){assert(earned.createdAt[root]<row.stageIndex);assert(earned.events.some(e=>e.out===root&&e.created&&e.stage<row.stageIndex));}}
  const final=earned.rows.find(r=>r.stage==='snack')!;assert(!final.readyAtEntry);assert.deepEqual(final.actions,[{type:'place',item:'supper-trolley',location:'dining-car'}]);assert(final.roots.includes('supper-trolley'));
  for(const id of ['bean-supper','juice','processed-stew-tin']){assert(final.priorCreations.includes(id));assert(Object.values(M.token(earned.finalState,'supper-trolley')!.slots).includes(id));assert(earned.events.some(e=>e.out==='supper-trolley'&&e.inputs.includes(id)&&e.stage<final.stageIndex));}
  const cakes=['vanilla-sandwich','iced-chocolate-cake'].filter(id=>Object.values(M.token(earned.finalState,'supper-trolley')!.slots).includes(id));assert(cakes.length);for(const id of cakes){assert(final.priorCreations.includes(id));assert(earned.events.some(e=>e.out==='supper-trolley'&&e.inputs.includes(id)&&e.stage<final.stageIndex));}
 });
}

test('accepted-goal ancestry rejects primitive-only cargo, missing install links and unrelated creations',()=>{
 const edges=new Map([['trolley',new Set(['raw-food'])]]),made=new Map([['cooked-food',0]]);
 assert.deepEqual(priorCreatedAncestors(['trolley'],edges,made,1),[],'unrelated cooked food must not count');
 assert.deepEqual(priorCreatedAncestors(['wrong-primitive'],new Map([['trolley',new Set(['cooked-food'])]]),made,1),[],'wrong goal witness must not count');
 assert.deepEqual(priorCreatedAncestors(['trolley'],new Map(),made,1),[],'inventory without accepted install link must not count');
 assert.deepEqual(priorCreatedAncestors(['trolley'],edges,new Map(),1),[],'a merely loaded primitive is not an earlier creation');
 assert.deepEqual(priorCreatedAncestors(['trolley'],new Map([['trolley',new Set(['cooked-food'])]]),made,0),[],'same-stage manufacture is not earlier-chapter creation');
 assert.deepEqual(priorCreatedAncestors(['trolley'],new Map([['trolley',new Set(['cooked-food'])]]),made,1),['cooked-food']);
});

import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import * as M from './engine/model.ts';
import {validateConstructionContinuity,validateContentLifecycles,validateObjectiveFlow} from './schema/lifecycle.ts';
import type {Scenario,Action} from './schema/scenario.ts';
const read=(n:string)=>JSON.parse(readFileSync(new URL(n,import.meta.url),'utf8'));
const save=(n:string,v:unknown)=>writeFileSync(new URL(n,import.meta.url),JSON.stringify(v,null,2)+'\n');
const source=process.argv[2]??'cargo-alternative.candidate.scenario.json';
const prefix=source.startsWith('recommended.')?'recommended':'cargo-alternative';
const s:Scenario=read(source);
const checks:any[]=[];const check=(name:string,fn:()=>unknown)=>checks.push({name,passed:true,evidence:fn()});
const snapshot=(g:M.Game)=>({raw:g.state.tokens.filter(t=>['cushioned-capsule','specimen-cart','loaded-capsule-dock','loaded-ascent-vehicle','flight-ready-vehicle','ascending-vehicle','docked-vehicle'].includes(t.kind)),visible:M.playerView(s,g).things.filter(t=>['cushioned-capsule','specimen-cart','loaded-capsule-dock','loaded-ascent-vehicle','flight-ready-vehicle','ascending-vehicle','docked-vehicle'].includes(t.id))});
const owners=(g:M.Game,kind:string)=>g.state.tokens.filter(t=>t.kind===kind||Object.values(t.slots).includes(kind)||t.components?.includes(kind));
function trace(actions:Action[]){let game=M.createGame(s);const points=new Map<string,{before:M.Game,after:M.Game}>(),transfers:any[]=[],stages:any[]=[];
 for(const [index,action] of actions.entries()){const before=game,result=M.applyAction(s,game,action);assert.ok(result.accepted,`Action ${index+1}: ${JSON.stringify(action)}: ${result.message}`);game=result.game;
  if(result.recipe){points.set(result.recipe,{before,after:game});if(result.recipe==='departure-11-specimen-cart'||result.recipe.startsWith('homebound-'))stages.push({acceptedActionIndex:index+1,recipe:result.recipe,action,retired:result.retired??[],after:snapshot(game)});}
  if(action.type==='advance'&&s.stages[game.state.stage].id==='homebound'&&game.state.stage!==before.state.stage)transfers.push(...game.events.at(-1)!.transfers??[]);
 }assert.ok(game.state.completed);return {game,points,stages,transfers};}
check('Existing scenario, lifecycle, objective-flow and construction validators pass',()=>{const errors=[...M.validateScenario(s),...validateContentLifecycles([s]),...validateObjectiveFlow([s]),...validateConstructionContinuity([s])];assert.deepEqual(errors,[]);return errors;});
const runs=new Map<string,ReturnType<typeof trace>>();
check('All five prior route sequences complete with only load input and final gesture changed; one fewer synthetic combination',()=>{
 const out=[];for(const name of ['authored','beacon','tow','tow-old-alias','closed-reader.accepted']){const original:Action[]=read(`${name}.actions.json`),actions:Action[]=read(`${name}.cargo.actions.json`);assert.equal(actions.length,original.length);const diffs=actions.flatMap((a,i)=>JSON.stringify(a)!==JSON.stringify(original[i])?[{index:i+1,before:original[i],after:a}]:[]);assert.equal(diffs.length,2);assert.ok(JSON.stringify(diffs[0].after).includes('cushioned-capsule'));assert.deepEqual(diffs[1].after,{type:'place',item:'docked-vehicle',location:'orbital-hatch'});const run=trace(actions);runs.set(name,run);out.push({name,actions:actions.length,combinations:run.game.state.metrics.combinations,completed:true,diffs});}assert.equal(runs.get('closed-reader.accepted')!.game.state.metrics.combinations,203);return out;
});
const run=runs.get('closed-reader.accepted')!;
check('Cart preparation keeps precisely one filled capsule and does not snapshot any of its cargo',()=>{
 const point=run.points.get('departure-11-specimen-cart')!,cap=M.token(point.before.state,'cushioned-capsule')!,cart=M.token(point.after.state,'specimen-cart')!;
 assert.deepEqual(M.token(point.after.state,'cushioned-capsule'),cap);assert.equal(Object.keys(cap.slots).length,5);assert.deepEqual(owners(point.after,'cushioned-capsule').map(t=>t.id),['cushioned-capsule']);for(const k of ['cushioned-capsule',...Object.values(cap.slots),...cap.components??[]])assert.ok(!cart.components?.includes(k));assert.ok(cart.components?.includes('hauler-platform'));return snapshot(point.after);
});
check('Homebound carries prepared cart and capsule separately, plus cabin, record and rendezvous',()=>{for(const id of ['specimen-cart','cushioned-capsule','ascent-cabin','cabin-test-record','agreed-rendezvous'])assert.ok(run.transfers.some(x=>x.item===id));return run.transfers;});
check('Actual capsule transfer removes capsule once, preserves all cargo, leaves the empty hauler untouched',()=>{
 const p=run.points.get('homebound-5-loaded-capsule-dock')!,cap=M.token(p.before.state,'cushioned-capsule')!,cart=M.token(p.before.state,'specimen-cart')!,dock=M.token(p.after.state,'loaded-capsule-dock')!;assert.equal(M.token(p.after.state,'cushioned-capsule'),undefined);assert.deepEqual(M.token(p.after.state,'specimen-cart'),cart);assert.deepEqual(owners(p.after,'cushioned-capsule').map(t=>t.id),['loaded-capsule-dock']);for(const k of ['cushioned-capsule',...Object.values(cap.slots),...cap.components??[]])assert.ok(dock.components?.includes(k));for(const k of cart.components??[])assert.ok(!dock.components?.includes(k));return snapshot(p.after);
});
check('Every flight transformation has exactly one capsule owner, all same contents, no hauler aboard',()=>{
 const packed=run.points.get('homebound-5-loaded-capsule-dock')!.before,cap=M.token(packed.state,'cushioned-capsule')!,hauler=M.token(packed.state,'specimen-cart')!;
 const evidence=[];for(const id of ['homebound-6-loaded-ascent-vehicle','homebound-9-flight-ready-vehicle','homebound-11-ascending-vehicle','homebound-12-docked-vehicle']){const p=run.points.get(id)!,owner=owners(p.after,'cushioned-capsule');assert.equal(owner.length,1);for(const k of ['cushioned-capsule',...Object.values(cap.slots),...cap.components??[]])assert.ok(owner[0].components?.includes(k));for(const k of ['specimen-cart',...hauler.components??[]])assert.ok(!owner[0].components?.includes(k));assert.deepEqual(M.token(p.after.state,'specimen-cart'),hauler);evidence.push({recipe:id,capsuleOwner:owner[0].id,contents:owner[0].components});}return evidence;
});
check('Missing capsule compartment blocks both fitting and load, in both drag directions',()=>{
 const evidence=[];for(const id of ['departure-11-specimen-cart','homebound-5-loaded-capsule-dock']){const p=run.points.get(id)!,recipe=s.recipes.find(x=>x.id===id)!;for(const slot of Object.keys(M.token(p.before.state,'cushioned-capsule')!.slots)){for(const [a,b] of [[recipe.a,recipe.b],[recipe.b,recipe.a]]){const game=structuredClone(p.before);delete M.token(game.state,'cushioned-capsule')!.slots[slot];const result=M.applyAction(s,game,{type:'combine',a,b});assert.equal(result.accepted,false);evidence.push({recipe:id,missingSlot:slot,direction:[a,b],accepted:false});}}}return evidence;
});
check('Both operand directions perform the same single-owner transfer',()=>{const evidence=[];for(const id of ['departure-11-specimen-cart','homebound-5-loaded-capsule-dock','homebound-6-loaded-ascent-vehicle']){const before=run.points.get(id)!.before,r=s.recipes.find(x=>x.id===id)!;const x=M.applyAction(s,before,{type:'combine',a:r.a,b:r.b}),y=M.applyAction(s,before,{type:'combine',a:r.b,b:r.a});assert.ok(x.accepted&&y.accepted);assert.deepEqual(snapshot(x.game),snapshot(y.game));evidence.push({recipe:id,sameStateAndVisibleContents:true});}return evidence;});
check('Final placement wins with real cargo visible on docked ship and creates no capsule badge',()=>{const view=M.playerView(s,run.game),ship=view.things.find(x=>x.id==='docked-vehicle')!;assert.equal(ship.location,'orbital-hatch');assert.ok(ship.gold);assert.equal(M.token(run.game.state,'returned-capsule'),undefined);assert.equal(s.recipes.find(r=>r.id==='homebound-13-reunited-crew'),undefined);assert.ok(ship.components.includes('Catalogued ice sample'));assert.ok(ship.components.includes('Complete field archive'));return {completed:run.game.state.completed,goals:view.objectives,ship,cart:view.things.find(x=>x.id==='specimen-cart'),completionText:view.completionText};});
check('Undo restores single prior capsule owner; JSON restore retains final ownership',()=>{const before=run.points.get('homebound-5-loaded-capsule-dock')!.before,after=run.points.get('homebound-5-loaded-capsule-dock')!.after,undone=M.undo(after);assert.ok(undone.accepted);assert.deepEqual(snapshot(undone.game),snapshot(before));const restored=M.restore(s,JSON.stringify(run.game));assert.deepEqual(snapshot(restored),snapshot(run.game));return {undoMatchesBefore:true,restoreMatchesAfter:true};});
check('Every embedded checkpoint still replays on unchanged engine',()=>{const results=[];for(const c of s.checkpoints??[]){const game=M.checkpoint(s,c.id);results.push({id:c.id,actions:c.actions.length,stage:s.stages[game.state.stage].id,completed:game.state.completed});}return results;});
save(`${prefix}.verification.json`,{evidenceClass:'Source-aware logical candidate verification, not a new blind playtest; no UI/browser testing',checks});save(`${prefix}.trace.json`,{metrics:run.game.state.metrics,transfers:run.transfers,stages:run.stages,final:snapshot(run.game)});console.log(JSON.stringify({passed:checks.length,failed:0,checks:checks.map(x=>x.name)},null,2));

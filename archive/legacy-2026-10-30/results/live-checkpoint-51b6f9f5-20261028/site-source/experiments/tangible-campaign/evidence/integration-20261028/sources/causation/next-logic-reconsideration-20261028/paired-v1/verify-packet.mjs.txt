import assert from 'node:assert/strict';
import {readFileSync,readdirSync,lstatSync,existsSync,writeFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {resolve,relative} from 'node:path';
import {pathToFileURL} from 'node:url';
const sha=b=>createHash('sha256').update(b).digest('hex');
const r=JSON.parse(readFileSync(new URL('./evidence/reader-packet.json',import.meta.url),'utf8'));
const base=r.packet,bytes=readFileSync(resolve(base,'manifest.json'));assert.equal(sha(bytes),r.manifestSha256);
assert.equal(readFileSync(resolve(base,'MANIFEST-SHA256.txt'),'utf8').trim(),r.manifestSha256);
const manifest=JSON.parse(bytes);assert.equal(manifest.runtimeCommit,'61a9b4ee8c6235bff07f97eb16d713ec6125c81e');
for(const [name,h] of Object.entries(manifest.files)){
 const file=resolve(base,name);assert(file.startsWith(base+'/'));assert(lstatSync(file).isFile());assert(!lstatSync(file).isSymbolicLink());
 assert.equal(sha(readFileSync(file)),h,name);assert.equal(lstatSync(file).mode&0o222,0);
}
const files=[];const walk=dir=>{for(const n of readdirSync(dir)){const f=resolve(dir,n);if(lstatSync(f).isDirectory())walk(f);else files.push(relative(base,f));}};walk(base);
assert.deepEqual(files.sort(),[...Object.keys(manifest.files),'manifest.json','MANIFEST-SHA256.txt'].sort());
assert.equal(sha(readFileSync(new URL('./scenario.json',import.meta.url))),r.scenarioSha256);
assert.equal(sha(readFileSync(resolve(base,'scenario.json'))),r.scenarioSha256);
const task=readFileSync(resolve(base,'PLAYER_TASK.md'),'utf8');for(const word of ['Aristotle','causation','reflected','mirror','matched','checkpoint'])assert(!task.includes(word),word);
assert(!existsSync(resolve('/workspace/shared/sessions',base.split('/').at(-1))));
const M=await import(pathToFileURL(resolve(base,'engine/model.ts'))),S=await import(pathToFileURL(resolve(base,'surface.mjs')));
const scenario=JSON.parse(readFileSync(resolve(base,'scenario.json'),'utf8')),g=M.createGame(scenario),v=S.visibleView(M,scenario,g);
assert.equal(g.events.length,0);assert.equal(g.state.stage,0);assert.equal(v.completed,false);assert.deepEqual(v.things.map(t=>t.id),['papers']);assert(!Object.hasOwn(v.scenario,'id'));
const checkpoints=scenario.checkpoints.map(c=>{const x=M.checkpoint(scenario,c.id);return {id:c.id,label:c.label,authoredActions:c.actions,gameSha256:sha(JSON.stringify(x)),stage:x.state.stage+1,metrics:x.state.metrics,completed:x.state.completed,game:x};});
writeFileSync(new URL('./evidence/earned-checkpoints.json',import.meta.url),JSON.stringify({evidenceClass:'source-aware replay-earned checkpoint evidence; not reader actions',checkpoints},null,2)+'\n');
const result={...r,evidenceClass:'source-aware static packet verification and zero-action in-memory opening; no reader start',filesVerified:Object.keys(manifest.files).length,zeroActionOpeningVerified:true,earnedCheckpointsVerified:checkpoints.length,readerSessionDirectoryAbsent:true};
writeFileSync(new URL('./evidence/packet-verification.json',import.meta.url),JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result,null,2));

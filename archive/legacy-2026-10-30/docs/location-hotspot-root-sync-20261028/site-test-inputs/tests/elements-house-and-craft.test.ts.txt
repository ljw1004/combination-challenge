const route=s.checkpoints!.find(cp=>cp.id==='complete')!.actions;
const c=(a:string,b:string):Action=>({type:'combine',a,b});
const pair=(a:string,b:string)=>[a,b].sort().join('|');
const act=(g:M.Game,a:Action)=>{const r=M.applyAction(s,g,a);assert(r.accepted,`${g.state.stage+1} ${JSON.stringify(a)}: ${r.message}`);return r;};
const output=new URL('../results/elements-house-and-craft-v1/',import.meta.url);fs.mkdirSync(output,{recursive:true});
function run(actions:Action[]) {
  let g=M.createGame(s);const createdAt=new Map<string,number>(),created:string[]=[];
  const rows=s.stages.map(st=>({id:st.id,connected:[] as string[]}));let edges=new Map<string,Set<string>>();
  for(const a of actions){const stage=g.state.stage,r=act(g,a);for(const t of g.state.tokens)if(!M.token(r.game.state,t.id))assert(a.type==='combine'&&[a.a,a.b].includes(t.id),`unrelated removal ${t.id}`);
    if(a.type==='combine'){const out=r.created??r.updated;if(out){const deps=edges.get(out)??new Set<string>();for(const k of [a.a,a.b].filter(k=>!k.startsWith('@')))deps.add(k);edges.set(out,deps);}if(r.created){createdAt.set(r.created,stage);created.push(r.created);}}
    if(a.type==='advance'){const connected=new Set<string>(),visited=new Set<string>();const walk=(id:string)=>{if(visited.has(id))return;visited.add(id);for(const k of edges.get(id)??[]){if(createdAt.has(k)&&createdAt.get(k)!<stage)connected.add(k);if(k!==id)walk(k);}};for(const o of M.objectiveStates(s,g.state))for(const id of o.witnesses.length?o.witnesses:o.intermediate&&o.item?[o.item]:[])walk(id);rows[stage].connected=[...connected];edges=new Map();}
    g=r.game;g.history=[];
  }return {game:g,created,rows};
}
function swap(actions:Action[],a:string,b:string,replacement:Action[]){let n=0;const out=actions.flatMap(x=>{if(x.type==='combine'&&pair(x.a,x.b)===pair(a,b)){n++;return replacement;}return [x];});assert.equal(n,1,`${a}+${b}`);return out;}
function glassRoute(actions:Action[]){
  const deferred=new Map<string,Action>([['glaze',c('sand','clay')],['glazed-bottle',c('jar','glaze')]]),out:Action[]=[];
  const supply=(k:string)=>{const a=deferred.get(k);if(!a)return;deferred.delete(k);if(a.type==='combine')for(const input of [a.a,a.b])supply(input);out.push(a);};
  for(const a of actions){if(a.type==='combine'&&[pair('sand','clay'),pair('jar','glaze')].includes(pair(a.a,a.b)))continue;
    if(a.type==='combine'&&pair(a.a,a.b)===pair('cider','glazed-bottle')){out.push(c('glass','air'),c('cider','glass-bottle'));continue;}
    for(const k of a.type==='combine'?[a.a,a.b]:a.type==='move'||a.type==='place'?[a.item]:[])supply(k);out.push(a);
  }return out;
}
function directOre(actions:Action[]){return swap(swap(actions,'hammer','@quarry',[]),'stone','water',[c('stone','@quarry'),c('stone','water')]);}

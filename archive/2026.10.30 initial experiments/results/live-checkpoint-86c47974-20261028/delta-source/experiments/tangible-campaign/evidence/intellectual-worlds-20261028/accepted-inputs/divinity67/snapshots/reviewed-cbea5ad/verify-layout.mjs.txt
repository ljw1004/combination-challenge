/** Computed geometry under the user's contract: compact birth overlap is allowed; transport must not add new overlaps. No UI claims. */
import fs from 'node:fs';import crypto from 'node:crypto';import assert from 'node:assert/strict';import * as M from './runtime/experiments/tangible-campaign/engine/model.ts';
const P=new URL('.',import.meta.url),bytes=fs.readFileSync(new URL('conceptual-candidate.json',P)),s=JSON.parse(bytes),birthOverlaps=[],arrivalRegressions=[],otherNewOverlaps=[],observations=[];
const key=(a,b)=>[a.id,b.id].sort().join('|');
const overlaps=state=>{const out=new Map();for(let i=0;i<state.tokens.length;i++)for(let j=i+1;j<state.tokens.length;j++){const a=state.tokens[i],b=state.tokens[j];if(a.area!==b.area)continue;const ha=M.tokenHeight(s,a),hb=M.tokenHeight(s,b),dx=96-Math.abs(a.x-b.x),dy=(ha+hb)/2-Math.abs(a.y-b.y);if(dx>0.5&&dy>0.5)out.set(key(a,b),{a:a.id,b:b.id,area:a.area,dx,dy,geometricContainment:Math.abs(a.x-b.x)<.001&&Math.abs(a.y-b.y)<=Math.abs(ha-hb)/2});}return out;};
for(const width of [320,600])for(const reversed of [false,true]){let g=M.createGame(s,width),event=0;assert.equal(overlaps(g.state).size,0,'Initial source delivery must be non-overlapping.');
 for(let action of s.checkpoints.at(-1).actions){const before=overlaps(g.state);if(reversed&&action.type==='combine')action={...action,a:action.b,b:action.a};const r=M.applyAction(s,g,action);assert(r.accepted);g=r.game;event++;const after=overlaps(g.state),fresh=[...after].filter(([k])=>!before.has(k)).map(([,x])=>({width,reversed,event,stage:g.state.stage+1,action:action.type,...x}));observations.push({width,reversed,event,activeStage:g.state.stage+1,newOverlapPairs:fresh.length});
  if(action.type==='advance')arrivalRegressions.push(...fresh);
  else for(const pair of fresh)if([pair.a,pair.b].includes(r.created))birthOverlaps.push(pair);else otherNewOverlaps.push(pair);
 }
}

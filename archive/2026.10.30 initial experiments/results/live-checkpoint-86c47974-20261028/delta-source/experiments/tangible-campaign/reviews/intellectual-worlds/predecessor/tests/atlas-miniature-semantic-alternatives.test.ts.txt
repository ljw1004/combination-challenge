// Logical/data-only integration checks. No DOM/controller/browser imports.
import test from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs';
import * as M from '../experiments/tangible-campaign/engine/model.ts';
import {scenario as s} from '../experiments/tangible-campaign/ui/campaigns/atlas-miniature-research.ts';
import {scenario as original} from '../experiments/tangible-campaign/ui/campaigns/atlas.ts';
import {featuredScenarioIds,scenarios} from '../experiments/tangible-campaign/ui/scenarios.ts';
const route=JSON.parse(fs.readFileSync(new URL('../experiments/tangible-campaign/evidence/atlas-three-philosophical-cases-v2-research-route-evidence.json',import.meta.url),'utf8'));
function entries(){let g=M.createGame(s);const out=new Map<number,typeof g>();for(const a of route.actionsLog){const x=M.applyAction(s,g,a);assert(x.accepted);g=x.game;if(a.type==='advance'&&!out.has(g.state.stage))out.set(g.state.stage,structuredClone(g))}return out}
function combine(g:ReturnType<typeof M.createGame>,a:string,b:string){const x=M.applyAction(s,g,{type:'combine',a,b});assert(x.accepted,x.message);return x.game}
test('Atlas miniature remains optional; original full Atlas remains a preserved snapshot',()=>{assert.equal(featuredScenarioIds.length,6);assert(!featuredScenarioIds.includes(original.id));assert(featuredScenarioIds.includes('atlas-connected-questions-v9-evidence-repair-research'));assert(!featuredScenarioIds.includes(s.id));assert(scenarios.some(x=>x.id===s.id));assert.equal(s.items.length,57);assert.equal(s.recipes.length,32);assert.equal(s.stages.length,4);assert.deepEqual(s.stages[0],original.stages[0]);});
test('Atlas miniature checkpoint earns the exact original Babylon opening',()=>{const cp=s.checkpoints!.find(x=>x.id==='after-babylon-watch')!;assert(cp);assert.equal(cp.actions.filter(x=>x.type==='combine').length,7);assert.equal(cp.actions.at(-1)!.type,'advance');const before=cp.actions.slice(0,-1);assert.deepEqual(M.replay(s,before),{...M.replay(original,before),scenarioId:s.id});const g=M.checkpoint(s,cp.id);assert.equal(g.state.stage,1);assert.equal(g.state.metrics.combinations,7);assert.deepEqual(M.restore(s,JSON.stringify(g)),g);});
test('Atlas miniature supports direct case contrast without consensus',()=>{for(const [a,b] of [['p-physician','p-shepherd'],['p-shepherd','p-physician']]){const entry=entries().get(2)!;const g=combine(entry,a,b);assert(g.state.discovered.includes('p-craft-rival'));assert(g.state.tokens.some(t=>t.kind==='p-expert-ruler'));assert(!M.stageComplete(s,g.state));}});
test('Atlas miniature help does not require the optional Cynic intermediate',()=>{for(const [a,b] of [['p-judgment-space','p-hungry-citizen'],['p-hungry-citizen','p-judgment-space']]){const entry=entries().get(3)!;entry.state.tokens=entry.state.tokens.filter(t=>t.kind!=='p-independent-need');let g=combine(entry,'p-insult','p-judgment-practice');g=combine(g,a,b);assert(g.state.discovered.includes('p-help-despite-insult'));assert(!g.state.discovered.includes('p-different-needs'));assert(!M.stageComplete(s,g.state));g=combine(g,'p-return-edict','p-return-qualified');g=combine(g,'p-edict-counterexample','p-correction-practice');assert(M.stageComplete(s,g.state));assert.deepEqual(M.restore(s,JSON.stringify(g)),g);}});
